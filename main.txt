# ==============================
# TEXT TO SPEECH
# ==============================

def speak(text):

    # Keep the original answer visible in the terminal
    print(f"{BOT_NAME}: {text}")

    # Remove Markdown symbols before speaking
    clean_text = re.sub(r"[*_#`]", "", text)

    # Remove bullet formatting
    clean_text = re.sub(r"^\s*[-•]\s*", "", clean_text)

    # Clean extra spaces/newlines
    clean_text = re.sub(r"\s+", " ", clean_text).strip()

    try:

        # Generate speech and send it directly to PulseAudio/PipeWire
        speech = subprocess.Popen(
            ["espeak-ng", "-s", "150", "--stdout", clean_text],
            stdout=subprocess.PIPE
        )

        subprocess.run(
            [
                "paplay",
                "--device=bluez_output.AA_53_D4_B4_A2_C8.1"
            ],
            stdin=speech.stdout,
            check=True
        )

        speech.stdout.close()
        speech.wait()

    except Exception as e:

        print("TTS error:", e)

# ==============================
# LOAD ALL EXCEL DATABASES
# ==============================

def load_database():

    database = []

    # Check knowledge-base folder
    if not os.path.isdir(KNOWLEDGE_BASE_FOLDER):

        print()
        print(
            f"WARNING: Folder not found: "
            f"{KNOWLEDGE_BASE_FOLDER}"
        )

        return database

    # Find every Excel file
    excel_files = []

    for filename in os.listdir(KNOWLEDGE_BASE_FOLDER):

        if filename.lower().endswith(".xlsx"):

            filepath = os.path.join(
                KNOWLEDGE_BASE_FOLDER,
                filename
            )

            excel_files.append(filepath)

    # No Excel files
    if not excel_files:

        print()
        print(
            "WARNING: No Excel files found in "
            f"{KNOWLEDGE_BASE_FOLDER}"
        )

        return database

    print()
    print("=" * 50)
    print("LOADING KITS KNOWLEDGE BASE")
    print("=" * 50)

    # Load every Excel file
    for excel_file in sorted(excel_files):

        print(
            f"Loading: "
            f"{os.path.basename(excel_file)}"
        )

        try:

            workbook = openpyxl.load_workbook(
                excel_file,
                data_only=True
            )

            for sheet in workbook.worksheets:

                # Skip empty sheets
                if sheet.max_row < 2:
                    continue

                headers = []

                # First row contains column names
                for cell in sheet[1]:

                    if cell.value is not None:

                        headers.append(
                            str(cell.value).strip()
                        )

                    else:

                        headers.append("")

                # Read all records
                for row in sheet.iter_rows(
                    min_row=2,
                    values_only=True
                ):

                    item = {}

                    for i, value in enumerate(row):

                        if i >= len(headers):
                            continue

                        header = headers[i]

                        if not header:
                            continue

                        if value is not None:

                            item[header] = value

                    if item:

                        # Keep track of where the record came from
                        item["_source_file"] = (
                            os.path.basename(
                                excel_file
                            )
                        )

                        item["_source_sheet"] = (
                            sheet.title
                        )

                        database.append(item)

            workbook.close()

        except Exception as e:

            print(
                f"ERROR loading "
                f"{os.path.basename(excel_file)}:"
            )

            print(e)

    print()
    print(
        f"Database loaded: "
        f"{len(database)} records"
    )

    print(
        f"Excel files loaded: "
        f"{len(excel_files)}"
    )

    print("=" * 50)
    print()

    return database


DATABASE = load_database()


# ==============================
# NORMALIZE PC REFERENCES
# ==============================

def normalize_pc_text(text):

    text = str(text).lower()

    replacements = {

        "computer number": "pc",
        "computer no": "pc",
        "computer #": "pc",
        "computer": "pc",

        "machine number": "pc",
        "machine no": "pc",
        "machine #": "pc",
        "machine": "pc",

        "pc number": "pc",
        "pc no": "pc",
        "pc #": "pc",
    }

    for old, new in replacements.items():

        text = text.replace(
            old,
            new
        )

    # PC 4 -> PC 04
    # PC 04 -> PC 04
    text = re.sub(
        r"\bpc[\s_-]*0*(\d+)\b",
        lambda match:
        f"pc {int(match.group(1)):02d}",
        text
    )

    return text


# ==============================
# SEARCH DATABASE
# ==============================

def search_database(question):

    question_lower = question.lower().strip()

    normalized_question = normalize_pc_text(
        question_lower
    )

    results = []

    # --------------------------------
    # CHECK FOR SPECIFIC PC
    # --------------------------------

    pc_match = re.search(
        r"\bpc\s+(\d+)\b",
        normalized_question
    )

    if pc_match:

        requested_pc = int(
            pc_match.group(1)
        )

        print(
            f"Database search: "
            f"looking for PC "
            f"{requested_pc:02d}"
        )

        pc_results = []

        for item in DATABASE:

            searchable_text = " ".join(

                str(value).lower()

                for key, value in item.items()

                if not key.startswith("_")
                and value is not None
            )

            normalized_database = (
                normalize_pc_text(
                    searchable_text
                )
            )

            database_pc_match = re.search(
                r"\bpc\s+(\d+)\b",
                normalized_database
            )

            if database_pc_match:

                database_pc = int(
                    database_pc_match.group(1)
                )

                if database_pc == requested_pc:

                    pc_results.append(item)

        if pc_results:

            print(
                f"Database match found: "
                f"PC {requested_pc:02d}"
            )

            return pc_results

        print(
            f"No direct PC "
            f"{requested_pc:02d} match found."
        )

    # --------------------------------
    # GENERAL SEARCH
    # --------------------------------

    ignored_words = {

        "what",
        "whats",
        "what's",
        "is",
        "are",
        "the",
        "of",
        "for",
        "about",
        "tell",
        "me",
        "give",
        "please",
        "can",
        "you",
        "could",
        "would",
        "do",
        "does",
        "have",
        "has",
        "our",
        "we",
        "i",
        "know",
        "information",
        "specifications",
        "specification",
        "details",
        "detail",
        "show",
        "list",
        "tell"
    }

    words = re.findall(
        r"[a-zA-Z0-9]+",
        normalized_question
    )

    question_words = [

        word

        for word in words

        if len(word) > 2
        and word not in ignored_words
    ]

    for item in DATABASE:

        score = 0

        # Search every field
        for key, value in item.items():

            if key.startswith("_"):
                continue

            if value is None:
                continue

            key_text = str(
                key
            ).lower()

            value_text = str(
                value
            ).lower()

            # Column name match
            for word in question_words:

                if word in key_text:

                    score += 3

            # Value match
            for word in question_words:

                if word in value_text:

                    score += 2

        # Keywords column gets extra importance
        if "Keywords" in item:

            keywords = str(
                item["Keywords"]
            ).lower()

            if keywords:

                for keyword in re.split(
                    r"[,;]",
                    keywords
                ):

                    keyword = keyword.strip()

                    if keyword and keyword in question_lower:

                        score += 20

        if score > 0:

            results.append(
                (score, item)
            )

    # Highest matches first
    results.sort(
        key=lambda x: x[0],
        reverse=True
    )

    matched_records = [

        item

        for score, item in results[:10]
    ]

    print(
        f"Database search found "
        f"{len(matched_records)} "
        f"matching records."
    )

    return matched_records


# ==============================
# VOICE TO TEXT
# ==============================

def listen():

    wav_file = tempfile.mktemp(
        suffix=".wav"
    )

    try:

        print()
        print(
            "Listening for 5 seconds... "
            "Speak now!"
        )

        subprocess.run(

            [
                "arecord",

                "-D",
                MIC_DEVICE,

                "-f",
                "S16_LE",

                "-r",
                str(SAMPLE_RATE),

                "-c",
                str(CHANNELS),

                "-t",
                "wav",

                "-d",
                "5",

                "-q",

                wav_file
            ],

            check=True
        )

        print(
            "Recording finished."
        )

        print(
            "Converting speech to text..."
        )

        recognizer = sr.Recognizer()

        with sr.AudioFile(
            wav_file
        ) as source:

            audio = recognizer.record(
                source
            )

        try:

            text = recognizer.recognize_google(
                audio
            )

            print(
                "You:",
                text
            )

            return text

        except sr.UnknownValueError:

            print(
                "I couldn't understand "
                "what you said."
            )

            return None

        except sr.RequestError as e:

            print(
                "Speech recognition "
                f"service error: {e}"
            )

            return None

    except Exception as e:

        print(
            "Microphone error:",
            e
        )

        return None

    finally:

        if os.path.exists(
            wav_file
        ):

            os.remove(
                wav_file
            )

# ASK GEMINI
# ==============================

def ask_gemini(question):

    database_results = search_database(
        question
    )

    if database_results:

        database_text = (
            "\nAUTHORITATIVE "
            "STUDIO DATABASE:\n"
        )

        for number, item in enumerate(
            database_results,
            start=1
        ):

            database_text += (
                f"\n--- Record {number} ---\n"
            )

            for key, value in item.items():

                # Don't send internal fields
                if key.startswith("_"):
                    continue

                if value is not None:

                    database_text += (
                        f"{key}: {value}\n"
                    )

    else:

        database_text = (
            "\nNo matching information "
            "was found in the database.\n"
        )

    prompt = f"""
You are {BOT_NAME}, the AI assistant
for KITS studio.

Answer the user's question clearly
and naturally.

IMPORTANT DATABASE RULES:

1. The Excel database is the
   authoritative source for studio
   equipment information.

2. Use the database whenever it
   contains the answer.

3. Do NOT invent specifications.

4. Do NOT assume that information
   about one machine applies to
   another machine.

5. PC 4, PC 04, computer 4,
   computer number 4, and machine 4
   refer to the same PC.

6. If the database contains the
   requested information, answer
   directly.

7. If the database does not contain
   the information, clearly say that
   it is not available in the studio
   database.

8. You can combine information from
   multiple database records if
   necessary.

9. Do not mention Excel filenames
   unless the user asks.

DATABASE:
{database_text}

USER QUESTION:
{question}
"""

    response = client.models.generate_content(

        model="gemini-3.6-flash",

        contents=prompt
    )

    return response.text
